1. Compute $\kappa_{n}$ for $$\boldsymbol{\sigma}(u,v) = (u,v,u^{2}+v^{2})$$in the direction of $\gamma(t) = \boldsymbol{\sigma}(t^{2},t)$ at $\gamma(1)$.

Sol)
$$
\begin{align}
\boldsymbol{\sigma}_{u} &= (1, 0, 2u) \\
\boldsymbol{\sigma}_{v} &= (0, 1, 2v) \\
\boldsymbol{\sigma}_{u}\times \boldsymbol{\sigma}_{v} &= (-2u, -2v, 1) \\
\lVert \boldsymbol{\sigma}_{u}\times \boldsymbol{\sigma}_{v} \rVert &= \sqrt{ 1+4u^{2} +4v^{2}} \\
\boldsymbol{U}(u,v) &= \frac{1}{\sqrt{ 1+4u^{2} +4v^{2}}}(-2u, -2v, 1) \\ 
\end{align}
$$
$$
\begin{align}
\gamma'(t) &= \frac{ \partial \boldsymbol{\sigma} }{ \partial u }\frac{ du }{ dt } +\frac{ \partial \boldsymbol{\sigma} }{ \partial v } \frac{ dv }{ dt }  \\
&= 2t(1,0,2t^{2})+(0, 1,2t) \\
&= (2t, 1, 4t^{3}+2t) \\
\gamma''(t) &= (2, 0, 12t^{2}+2) \\
\gamma''(1)\cdot \boldsymbol{U}(1^{2}, 1) &= (2, 0, 14)\cdot \frac{1}{3}(-2,-2, 1) \\
&=-\frac{4}{3}-\frac{0}{3}+\frac{14}{3}=\frac{10}{3} \\
\gamma'(1) &= (2, 1, 6) \\
\lVert \gamma'(1) \rVert &= \sqrt{ 4+1+36 } =\sqrt{ 41 } \\
\kappa_{n} &= \frac{\gamma''(1)\cdot \boldsymbol{U}(1,1)}{\lVert \gamma'(1) \rVert }=\frac{10}{123 }
\end{align}
$$

2. Compute $\kappa_{n}$ for $$
\boldsymbol{\sigma}(u,v) = (u, v, u^{2}-v^{2})$$at $\boldsymbol{\sigma}(0,0)$.

Sol)
$$
\begin{align}
\boldsymbol{\sigma}_{u} &= (1, 0, 2u) \\
\boldsymbol{\sigma}_{v} &= (0, 1, -2v) \\
\boldsymbol{\sigma}_{u}\times \boldsymbol{\sigma}_{v} &= (-2u,2v, 1) \\
\boldsymbol{U}(0, 0) &= \frac{\boldsymbol{\sigma}_{u}\times \boldsymbol{\sigma}_{v}}{\lVert \boldsymbol{\sigma}_{u}\times \boldsymbol{\sigma}_{v} \rVert }=(0,0,1)
\end{align}
$$
$$
\begin{align}
L &= \boldsymbol{\sigma}_{uu}\cdot \boldsymbol{U} \\
&= (0,0,2)\cdot(0,0,1) = 2 \\
M &= \boldsymbol{\sigma}_{uv}\cdot \boldsymbol{U} \\
&= (0,0,0)\cdot(0,0,1) = 0 \\
N &= \boldsymbol{\sigma}_{vv}\cdot \boldsymbol{U} \\
&= (0, 0, -2)\cdot(0, 0, 1) = -2
\end{align}
$$
$$
\begin{align}
E &= \boldsymbol{\sigma}_{u}\cdot \boldsymbol{\sigma}_{u} \\
&= (1, 0, 0)\cdot(1, 0, 0) = 1 \\
F &= \boldsymbol{\sigma}_{u}\cdot \boldsymbol{\sigma}_{v} \\
&= (1, 0, 0)\cdot(0, 1, 0) = 0 \\
G &= \boldsymbol{\sigma}_{v}\cdot \boldsymbol{\sigma}_{v} \\
&= (0, 1, 0)\cdot(0, 1, 0) = 1
\end{align}
$$
$$
\begin{align}
\kappa_{n} &= \frac{\text{II}}{\text{I}} \\
&= \frac{\left( 2\left( \frac{ du }{ dt }  \right)^{2} - 2\left( \frac{ dv }{ dt }  \right)^{2}\right)}{\left( \frac{ du }{ dt }  \right)^{2}+\left( \frac{ dv }{ dt }  \right)^{2}}
\end{align}
$$

3. Compute $\kappa_{n}$ for $$
\boldsymbol{\sigma}(u,v) = (r\cos u\cos v, r\sin u\cos v, r\sin v)
$$ in the direction of $\gamma(t) = \boldsymbol{\sigma}(\frac{t}{\cos c},c)$ for some $-\frac{\pi}{2} < c < \frac{\pi}{2}$.

Sol)
$$
\begin{align}
E &= \boldsymbol{\sigma}_{u}\cdot \boldsymbol{\sigma}_{u} \\
&= (-r\sin u\cos v,r\cos u\cos v, 0)\cdot(-r\sin u\cos v, r \cos u\cos v, 0) \\
&= r^{2}\cos ^{2} v \\
F &= \boldsymbol{\sigma}_{u}\cdot \boldsymbol{\sigma}_{v} \\
&= (-r\sin u\cos v, r\cos u\cos v, 0)\cdot(-r\cos u\sin v, -r\sin u\sin v, r\cos v) \\
&= 0 \\
G &= (-r\cos u\sin v, -r\sin u\sin v, r\cos v)\cdot(-r\cos u\sin v, -r\sin u\sin v, r\cos v) \\
&= r^{2}\cos ^{2}v + r^{2}\sin ^{2}v = r^{2}
\end{align}
$$
$$
\begin{align}
\text{I} &= E\left( \frac{ du }{ dt }  \right)^{2} + 2F\left( \frac{ du }{ dt }  \right)\left( \frac{ dv }{ dt }  \right) + G\left( \frac{ dv }{ dt }  \right)^{2} \\
&= \frac{r^{2}\cos ^{2} v}{\cos ^{2}c} = r^{2}
\end{align}
$$
$$
\begin{align} \\
\boldsymbol{\sigma}_{u} &= (-r\sin u\cos v,r\cos u\cos v, 0) \\
\boldsymbol{\sigma}_{v} &= (-r\cos u\sin v, -r\sin u\sin v, r\cos v) \\
\boldsymbol{\sigma}_{u}\times \boldsymbol{\sigma}_{v} &= (r^{2}\cos u\cos ^{2}v,r^{2}\sin u \cos ^{2}v,r^{2}\sin v\cos v)\\
\lVert \boldsymbol{\sigma}_{u}\times \boldsymbol{\sigma}_{v} \rVert &= \sqrt{ r^{4}\cos ^{4}v+r^{4}\sin ^{2}v\cos ^{2}v }  \\
&= r^{2}\cos c \\
\boldsymbol{U} &= \frac{\boldsymbol{\sigma}_{u}\times \boldsymbol{\sigma}_{v}}{\lVert \boldsymbol{\sigma}_{u}\times \boldsymbol{\sigma}_{v} \rVert } = (\cos u\cos c, \sin u\cos c, \sin c) \\
L &= \boldsymbol{\sigma}_{uu}\cdot \boldsymbol{U} \\
&= (-r\cos u\cos v, -r\sin u\cos v, 0)\cdot(\cos u\cos c, \sin u\cos c, \sin c) \\
&= (-r\cos u \cos c, -r\sin u\cos c, 0)\cdot(\cos u\cos c, \sin u\cos c, \sin c) \\
&= -r\cos ^{2}u\cos ^{2}c-r\sin ^{2}u\cos ^{2}c=-r\cos ^{2}c \\
M &=\boldsymbol{\sigma}_{uv}\cdot \boldsymbol{U} \\
&= (r\sin u\sin c,-r\cos u\sin c, 0)\cdot(\cos u\cos c, \sin u\cos c, \sin c) \\
&=0 \\
N &=\boldsymbol{\sigma}_{vv}\cdot \boldsymbol{U} \\
&= (-r\cos u\cos c, -r\sin u\cos c, -r\sin c)\cdot(\cos u\cos c, \sin u\cos c, \sin c) \\
&=-r\cos ^{2}u\cos ^{2}c-r\sin ^{2}u\cos ^{2}c-r\sin ^{2}c \\
&=-r\cos ^{2}c-r\sin ^{2}c=-r
\end{align}
$$
$$
\begin{align}
\text{II} &= L\left( \frac{ du }{ dt }  \right)^{2} + 2M\left( \frac{ du }{ dt }  \right)\left( \frac{ dv }{ dt }  \right) + N\left( \frac{ dv }{ dt }  \right)^{2} \\
&= -r
\end{align}
$$
$$
\begin{align}
\kappa_{n} &= \frac{\text{II}}{\text{I}} = -\frac{1}{r}
\end{align}
$$

4. Compute $\kappa_{n}$ for $$
\boldsymbol{\sigma}(u,v)=(\cosh u\cos v,\cosh u\sin v,u)
$$in the direction of $\gamma_{1}(t)=\boldsymbol{\sigma}(u_{0},t)$ and $\gamma_{2}(t)=\boldsymbol{\sigma}(t,v_{0})$ at $\boldsymbol{\sigma}(u_{0},v_{0})$.

Sol)
Let's compute $\kappa_{n}$.
$$
\begin{align}
\frac{ du }{ dt } _{\gamma_{1}} &= 0 \\
\frac{ dv }{ dt } _{\gamma_{1}} &= 1 \\
\frac{ du }{ dt } _{\gamma_{2}} &= 1 \\
\frac{ dv }{ dt } _{\gamma_{2}} &= 0
\end{align}
$$
$$
\begin{align} 
\boldsymbol{\sigma}_{u} &= (\sinh u\cos v,\sinh u\sin v, 1) \\
\boldsymbol{\sigma}_{v} &= (-\cosh u\sin v, \cosh u\cos v, 0) \\
\boldsymbol{\sigma}_{u}\times \boldsymbol{\sigma}_{v} &= (-\cosh u\cos v,-\cosh u\sin v,\sinh u\cosh u) \\
\lVert \boldsymbol{\sigma}_{u}\times \boldsymbol{\sigma}_{v} \rVert &= \sqrt{ \cosh ^{2}u+\sinh ^{2}u\cosh ^{2}u } = \cosh ^{2} u \\
\boldsymbol{U} &= \frac{\boldsymbol{\sigma}_{u}\times \boldsymbol{\sigma}_{v}}{\lVert \boldsymbol{\sigma}_{u}\times \boldsymbol{\sigma}_{v} \rVert }=\left( -\frac{\cos v}{\cosh u},-\frac{\sin v}{\cosh u}, \frac{\sinh u}{\cosh u} \right) \\
\boldsymbol{\sigma}_{uu} &= (\cosh u\cos v,\cosh u\sin v,0) \\
\boldsymbol{\sigma}_{uv} &= (-\sinh u\sin v,\sinh u\cos v, 0) \\
\boldsymbol{\sigma}_{vv} &= (-\cosh u\cos v,-\cosh u\sin v, 0) \\
L&=\boldsymbol{\sigma}_{uu}\cdot \boldsymbol{U} \\
&=-\cos ^{2}v-\sin ^{2}v=-1 \\
M &= \boldsymbol{\sigma}_{uv}\cdot \boldsymbol{U} \\
&=0 \\
N &= \boldsymbol{\sigma}_{vv}\cdot \boldsymbol{U} \\
&= \cos ^{2}v+\sin ^{2}v = 1 \\
\text{II}_{\gamma_{1}} &= L\left( \frac{ du }{ dt }  \right)_{\gamma_{1}}^{2} + 2M\left( \frac{ du }{ dt }  \right)_{\gamma_{1}}\left( \frac{ dv }{ dt }  \right)_{\gamma_{1}}+N\left( \frac{ dv }{ dt }  \right)_{\gamma_{1}}^{2} \\
&= 1 \\
\text{II}_{\gamma_{2}} &= -1 \\
E &= \sinh ^{2}u+1 = \cosh ^{2}u \\
F &= 0 \\
G &= \cosh ^{2}u \\
\text{I}_{\gamma_{1}} &= E\left( \frac{ du }{ dt }  \right)_{\gamma_{1}}^{2} + 2F\left( \frac{ du }{ dt }  \right)_{\gamma_{1}}\left( \frac{ dv }{ dt }  \right)_{\gamma_{1}}+G\left( \frac{ dv }{ dt }  \right)_{\gamma_{1}}^{2} \\
&=G_{\gamma_{1}} = \cosh ^{2}u_{0} \\
\text{I}_{\gamma_{2}} &= E_{\gamma_{2}} = \cosh ^{2}t
\end{align}
$$
$$
\begin{align}
\kappa_{n_{\gamma_{1}}} &= \frac{1}{\cosh ^{2}u_{0}} \\
\kappa_{n_{\gamma_{2}}} &= -\frac{1}{\cosh ^{2}u_{0}}
\end{align}
$$

5. $\kappa_{n}\equiv 0$ for $$
\boldsymbol{\sigma}(u,v) = (u\cos v,u\sin v,\ln u)
$$ if and only if the considering curves $\gamma(t) = \boldsymbol{\sigma}(u(t),v(t))$ on it satisfy $$
\ln u(t) = \pm v(t) + c
$$ for some $c$.

Sol)
$$
\begin{align}
\boldsymbol{\sigma}_{u} &= \left( \cos v, \sin v, \frac{1}{u} \right) \\
\boldsymbol{\sigma}_{v} &= (-u\sin v, u\cos v, 0) \\
\boldsymbol{\sigma}_{u}\times \boldsymbol{\sigma}_{v} &= (-\cos v,-\sin v,u) \\
\lVert \boldsymbol{\sigma}_{u}\times \boldsymbol{\sigma}_{v} \rVert &=\sqrt{ 1+u^{2} } \\
\boldsymbol{\sigma}_{uu} &= \left( 0,0,-\frac{1}{u^{2}} \right) \\
\boldsymbol{\sigma}_{uv} &= (-\sin v,\cos v,0) \\
\boldsymbol{\sigma}_{vv} &= (-u\cos v,-u\sin v, 0) \\
\boldsymbol{U} &= \frac{\boldsymbol{\sigma}_{u}\times \boldsymbol{\sigma}_{v}}{\lVert \boldsymbol{\sigma}_{u}\times \boldsymbol{\sigma}_{v} \rVert } \\
&=\frac{1}{\sqrt{ 1+u^{2} }}(-\cos v,-\sin v,u) \\
E = \boldsymbol{\sigma}_{u}\cdot \boldsymbol{\sigma}_{u} &= 1+\frac{1}{u^{2}} \\
F = \boldsymbol{\sigma}_{u}\cdot \boldsymbol{\sigma}_{v} &= 0 \\
G = \boldsymbol{\sigma}_{v}\cdot \boldsymbol{\sigma}_{v} &= u^{2} \\
L = \boldsymbol{\sigma}_{uu}\cdot \boldsymbol{U} &=-\frac{1}{u\sqrt{ 1+u^{2} }} \\
M = \boldsymbol{\sigma}_{uv}\cdot \boldsymbol{U} &=0 \\
N = \boldsymbol{\sigma}_{vv}\cdot \boldsymbol{U} &= \frac{u}{\sqrt{ 1+u^{2} }}
\end{align}
$$
To be $\kappa_{n}\equiv 0$, $\text{II}$ must be $0$.
$$
\begin{align}
\text{II} = -\frac{1}{u\sqrt{ 1+u^{2} }}\left( \frac{ du }{ dt }  \right)^{2} &+ \frac{u}{\sqrt{ 1+u^{2} }}\left( \frac{ dv }{ dt }  \right)^{2} = 0 \\
\frac{1}{u^{2}}\left( \frac{ du }{ dt }  \right)^{2} &= \left( \frac{ dv }{ dt }  \right)^{2} \\
\frac{1}{u}\frac{ du }{ dt } &=\frac{ dv }{ dt }  \\
\rightarrow \ln u(t) &= \pm v(t) + c
\end{align}
$$
