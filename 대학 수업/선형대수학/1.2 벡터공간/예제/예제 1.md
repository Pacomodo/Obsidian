Recall : [[벡터공간]]
Field $F$에서 성분을 가져온 모든 $n$-tuple의 집합을 $\mathbf{F}^{n}$이라고 한다.
$\mathbf{u}=(a_{1},a_{2},\cdots, a_{n}), \mathbf{v} = (b_{1}, b_{2}, \cdots, b_{n}) \in \mathbf{F}^{n}, c \in F$일때, Addition과 Scalar multiplication을 다음과 같이 정의하면 $\mathbf{F}^{n}$은 $F$-vector space이다.
$$
\begin{align}
\mathbf{u}+\mathbf{v} &= (a_{1}+_{_{F}}b_{1}, a_{2}+_{_{F}}b_{2}, \cdots, a_{n}+_{_{F}}b_{n}) \\
c\cdot \mathbf{u} &= (c\cdot_{_{F}}a_{1}, c\cdot_{_{F}}a_{2}, \cdots, c\cdot_{_{F}}a_{n})
\end{align}
$$
___
1. 교환 법칙

$$
\begin{align}
\mathbf{u}+\mathbf{v} &= (a_{1}+_{_{F}}b_{1}, a_{2}+_{_{F}}b_{2}, \cdots, a_{n}+_{_{F}}b_{n}) \\
&=(b_{1}+_{_{F}}a_{1}, b_{2}+_{_{F}}a_{2}, \cdots, b_{n}+_{_{F}}a_{n}) \\
&=\mathbf{v} + \mathbf{u}
\end{align}
$$
2. 결합법칙

$$
\begin{align}
(\mathbf{u}+\mathbf{v})+\mathbf{w} &= (a_{1}+_{_{F}}b_{1}, \cdots, a_{n}+_{_{F}}b_{n})+(c_{1}, \cdots, c_{n}) \\
&=((a_{1}+_{_{F}}b_{1})+_{_{F}}c_{1}, \cdots, (a_{n}+_{_{F}}b_{n})+_{_{F}}c_{n}) \\
&=(a_{1}+_{_{F}}(b_{1}+_{_{F}}c_{1}), \cdots, a_{n}+_{_{F}}(b_{n}+_{_{F}}c_{n}))  \\
&= (a_{1}, \cdots, a_{n}) + (b_{1}+_{_{F}}c_{1},\cdots, b_{n}+_{_{F}}c_{n}) \\
&= \mathbf{u} + (\mathbf{v}+\mathbf{w})
\end{align}
$$

3. 항등원

$$
\begin{align}
(0_{_{F}}, \cdots, 0_{_{F}})+(a_{1}, \cdots, a_{n}) &= (0_{_{F}}+_{_{F}}a_{1}, \cdots, 0_{_{F}}+_{_{F}}a_{n}) \\
&= (a_{1}, \cdots, a_{n})
\end{align}
$$

4. 역원

$$
\begin{align}
(a_{1},\cdots, a_{n})+(-a_{1}, \cdots, -a_{n}) &= (a_{1}+_{_{F}}(-a_{1}), \cdots, a_{n}+_{_{F}}(-a_{n})) \\
&=(0_{_{F}}, \cdots, 0_{_{F}})
\end{align}
$$
5. Scalar multiplication and multiplicative identity for $F$

$$
\begin{align}
1_{_{F}}\cdot \mathbf{u} &= (1_{_{F}}\cdot_{_{F}}a_{1}, \cdots, 1_{_{F}}\cdot_{_{F}}a_{n}) \\
&=(a_{1}, \cdots, a_{n}) = \mathbf{u}
\end{align}
$$

6. Scalar multiplication and multiplication in $F$

$$
\begin{align}
(p\cdot_{_{F}}q)\cdot \mathbf{u} &= ((p\cdot_{_{F}}q)\cdot_{_{F}}a_{1}, \cdots, (p\cdot_{_{F}}q)\cdot_{_{F}}a_{n}) \\
&=(p\cdot_{_{F}}(q\cdot_{_{F}}a_{1}), \cdots. p\cdot_{_{F}}(q\cdot_{_{F}}a_{n})) \\
&= p\cdot(q\cdot \mathbf{u})
\end{align}
$$
7. Distributive law
$$
\begin{align}
c\cdot(\mathbf{u}+\mathbf{v}) &= c\cdot(a_{1}+_{_{F}}b_{1}, \cdots, a_{n}+_{_{F}}b_{n}) \\
&=(c\cdot_{_{F}}(a_{1}+_{_{F}}b_{1}),\cdots,c\cdot_{_{F}}(a_{n}+_{_{F}}b_{n})) \\
&=(c\cdot_{_{F}}a_{1} +_{_{F}}c\cdot_{_{F}}b_{1}, \cdots, c\cdot_{_{F}}a_{n} +_{_{F}}c\cdot_{_{F}}b_{n}) \\
&= (c\cdot_{_{F}}a_{1}, \cdots, c\cdot_{_{F}}a_{n})+(c\cdot_{_{F}}b_{1}, \cdots, c\cdot_{_{F}}b_{n}) \\
&= c\cdot \mathbf{u} + c\cdot \mathbf{v}
\end{align}
$$
8. Addition in $F$ and Scalar multiplication
$$
\begin{align}
(p+_{_{F}}q)\cdot \mathbf{u} &= ((p+_{_{F}}q)\cdot_{_{F}}a_{1}, \cdots, (p+_{_{F}}q)\cdot_{_{F}}a_{n}) \\
&= (p\cdot_{_{F}}a_{1}+_{_{F}}q\cdot_{_{F}}a_{1}, \cdots, p\cdot_{_{F}}a_{n}+_{_{F}}q\cdot_{_{F}}a_{n}) \\
&= (p\cdot_{_{F}}a_{1}, \cdots, p\cdot_{_{F}}a_{n}) + (q\cdot_{_{F}}a_{1}, \cdots, q\cdot_{_{F}}a_{n}) \\
&= p\cdot \mathbf{u} +q\cdot \mathbf{v}
\end{align}
$$
